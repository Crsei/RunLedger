JSON 在 SQLite 中均保存为 TEXT。下列是代码约定；DDL 没有通过 json_valid 或 JSON Schema 对这些列作通用验证。事件、检查点与轨迹记录不是同一个数据结构。

载体结构与语义源码
session_events.payload_json由 event_type 决定。对 ledger.*，保存 LedgerEntry：{ id, sessionId, parentId, timestamp, type, payload }；type 为 session / message / tool_call / tool_result / turn / agent_event / custom。其他事件不能一概解释为 LedgerEntry。src/runtime/session-runtime/sqlite-ledger.ts
src/runtime/ledger/types.ts
session_checkpoints.snapshot_jsonCheckpointSnapshot：{ cacheSchema: 1, boundary, sourceSequence, state: Record<string, unknown> }。读取时校验摘要、版本、边界与源序号；不合格则回退重放。src/runtime/session-runtime/checkpoint.ts
command_attempt_receipts.result_json可空的结果 JSON 字符串；关联 result_digest / evidence_digest。结构取决于命令结果，SQL 层未固定统一字段。src/storage/session-store/attempt-repository.ts
sessions.*locator_json工作区 locator 的 JSON；worktree_locator_json 与 source_workspace_locator_json 区分 worktree 和源工作区绑定。不是另一张 workspace 表。src/storage/session-store/catalog-repository.ts
trajectory_records.jsonIndexedTrajectoryRecord = TrajectoryRecord + 可选 sessionInputSeq / sessionOutputSeq / traceInput / traceOutput。详见下方结构清单。src/runtime/trajectory/index-store.ts

CREATE TABLE sessions (
  session_id TEXT PRIMARY KEY,
  workspace_id TEXT NOT NULL,
  repository_id TEXT NOT NULL,
  status TEXT NOT NULL,
  created_at_ms INTEGER NOT NULL,
  updated_at_ms INTEGER NOT NULL,
  head_sequence INTEGER NOT NULL DEFAULT 0,
  current_checkpoint_id TEXT,
  last_driver_client_id TEXT,
  driver_revision INTEGER NOT NULL DEFAULT 0,
  worktree_locator_json TEXT,
  settings_digest TEXT NOT NULL,
  title TEXT NULL,
  title_source TEXT NULL,
  title_updated_at_ms INTEGER NULL, source_workspace_locator_json TEXT, harness_profile_digest TEXT NOT NULL DEFAULT '377be8e8b88ac2f1f34122eb57592e300af62b375e6b6e01edc85d9d25de6238'
  CHECK (length(harness_profile_digest) = 64 AND harness_profile_digest NOT GLOB '*[^0-9a-f]*'), harness_profile_id TEXT NOT NULL DEFAULT 'standard' CHECK (harness_profile_id IN ('standard', 'minimal', 'plan')), harness_profile_version INTEGER NOT NULL DEFAULT 1 CHECK (typeof(harness_profile_version) = 'integer' AND harness_profile_version IN (1, 2)),
  CHECK (status IN ('active', 'recovery_required', 'paused', 'completed', 'failed', 'archived')),
  CHECK (title IS NULL OR (length(CAST(title AS BLOB)) BETWEEN 1 AND 160)),
  CHECK (title_source IS NULL OR title_source IN ('auto', 'user')),
  CHECK (title_updated_at_ms IS NULL OR (typeof(title_updated_at_ms) = 'integer' AND title_updated_at_ms >= 0)),
  CHECK (
    (title IS NULL AND title_source IS NULL AND title_updated_at_ms IS NULL)
    OR (title IS NOT NULL AND title_source IS NOT NULL AND title_updated_at_ms IS NOT NULL)
  )
);

sessions 的 4 个触发器：标题完整性与 Harness Profile 白名单

标题三列必须同为空或同为非空。Profile 白名单允许 standard@1、standard@2、minimal@1、minimal@2、plan@1，并校验精确 digest；版本与摘要是 Session 的持久化身份。

CREATE TRIGGER sessions_title_invariant_insert
BEFORE INSERT ON sessions
WHEN NOT ((NEW.title IS NULL AND NEW.title_source IS NULL AND NEW.title_updated_at_ms IS NULL)
       OR (NEW.title IS NOT NULL AND NEW.title_source IS NOT NULL AND NEW.title_updated_at_ms IS NOT NULL))
BEGIN
  SELECT RAISE(ABORT, 'sessions title state is incomplete');
END;

CREATE TRIGGER sessions_title_invariant_update
BEFORE UPDATE OF title, title_source, title_updated_at_ms ON sessions
WHEN NOT ((NEW.title IS NULL AND NEW.title_source IS NULL AND NEW.title_updated_at_ms IS NULL)
       OR (NEW.title IS NOT NULL AND NEW.title_source IS NOT NULL AND NEW.title_updated_at_ms IS NOT NULL))
BEGIN
  SELECT RAISE(ABORT, 'sessions title state is incomplete');
END;
